## 4.08
escribir un algoritmo que permita buscar, insertar o borrar un elemento
identificado con una clave determinada en una lista circular simplemente
encadenada.

```
accion ejercicio es
  ambiente
    objeto = registro
      clave: entero
      ..otros campos
    fin_registro

    nodo = registro
      obj: objeto
      prox: puntero a nodo
    fin_registro

    reg: objeto
    prim, p, a, q: puntero a nodo
    buscado, eleccion: entero

    procedimiento mostrar_menu() es
      escribir("opciones:")
      escribir("  [1] buscar un elemento")
      escribir("  [2] insertar un elemento")
      escribir("  [3] eliminar un elemento")
    fin_procedimiento

    procedimiento buscar() es
      p := prim
      si (prim = nil) entonces
        escribir("error: el objeto no se encuentra en la lista.")
      contrario
        mientras (*p.prox <> prim) y (*p.obj.clave <> buscado) hacer
          a := p
          p := *p.prox
        fin_mientras

        si (*p.prox = prim) y (*p.obj.clave <> buscado) entonces
          escribir("error: el objeto no se encuentra en la lista.")
        contrario
          escribir("el objeto solicitado es: ")
          ..mostrar registro campo por campo
        fin_si
      fin_si
    fin_procedimiento

    procedimiento insertar() es
      escribir("ingrese el objeto a insertar: ")
      reg.clave := buscado
      ..leer campo por campo

      nuevo(q)
      *q.obj := reg

      si (prim = nil) entonces
        prim := q
        *q.prox := q
      contrario
        p := prim
        a := nil
        mientras (*p.prox <> prim) y (*p.obj.clave < buscado) hacer
          a := p
          p := *p.prox
        fin_mientras

        si (*p.prox = prim) y (*p.obj.clave < buscado) entonces
          *p.prox := q
          *q.prox := prim
        contrario
          si (p = prim) entonces
            a := prim
            mientras (*a.prox <> prim) hacer
              a := *a.prox
            fin_mientras
            *a.prox := q
            *q.prox := prim
            prim := q
          contrario
            *a.prox := q
            *q.prox := p
          fin_si
        fin_si
      fin_si
    fin_procedimiento

    procedimiento eliminar() es
      p := prim
      a := nil
      si (prim = nil) entonces
        escribir("no se encontró el objeto a eliminar.")
      contrario
        mientras (*p.prox <> prim) y (*p.obj.clave <> buscado) hacer
          a := p
          p := *p.prox
        fin_mientras

        si (*p.prox = prim) y (*p.obj.clave <> buscado) entonces
          escribir("no se encontró el objeto a eliminar.")
        contrario
          si (p = prim) entonces
            si (*p.prox = prim) entonces
              prim := nil
            contrario
              a := prim
              mientras (*a.prox <> prim) hacer
                a := *a.prox
              fin_mientras
              prim := *p.prox
              *a.prox := prim
            fin_si
          contrario
            si (*p.prox = prim) entonces
              *a.prox := prim
            contrario
              *a.prox := *p.prox
            fin_si
          fin_si
          disponer(p)
        fin_si
      fin_si
    fin_procedimiento
  proceso
    p := prim

    mostrar_menu()
    escribir("ingrese la operación a realizar:")
    leer(eleccion)
    escribir("ingrese la clave del objeto: ")
    leer(buscado)

    segun eleccion hacer
      1: buscar()
      2: insertar()
      3: eliminar()
      contrario: escribir("error: opción inválida.")
    fin_segun
fin_accion
```
