## 4.15
supóngase que, utilizando una lista encadenada, se implementa una cola q
(estructura 'fifo', es decir, 'primero en entrar, primero en salir').
diseñar un algoritmo que permita insertar o extraer un elemento, comprobando
siempre si la cola está vacía o no.

```
accion ejercicio(prim, ult: puntero a nodo) es
  ambiente
    nodo = registro
      dato: entero
      prox: puntero a nodo
    fin_registro

    aux, q: puntero a nodo
    op: entero
    continuar: caracter

    procedimiento insertar() es
      nuevo(q)
      escribir("ingrese el número a almacenar en el nuevo elemento:")
      leer(*q.dato)
      *q.prox := nil
      si (prim = nil) entonces
        prim := q
      contrario
        *ult.prox := q
      fin_si
      ult := q
    fin_procedimiento

    procedimiento remover() es
      si (prim = nil) entonces
        escribir("no se encontraron elementos.")
      contrario
        aux := prim
        prim := *prim.prox
        si (prim = nil) entonces
          ult := nil
        fin_si
        disponer(aux)
      fin_si
    fin_procedimiento
  proceso
    aux := nil

    repetir
      escribir("acciones")
      escribir("  [1] insertar un elemento")
      escribir("  [2] remover un elemento")

      escribir("ingrese el número de la operación a realizar: ")
      repetir
        leer(op)
      hasta que op en 1..2

      segun op hacer
        1: insertar()
        2: remover()
      fin_segun

      escribir("desea realizar otra operación? [s/n]")
      leer(continuar)
    hasta que continuar = 'n'
fin_accion
```
