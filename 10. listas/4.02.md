## 4.02
se dispone de una lista simplemente encadenada de números enteros, diseñar un
algoritmo que a partir de ella genere otra lista conteniendo los nodos cuyos
datos terminan en cero; dichos elementos, deberán ser eliminados de la lista
original. se asume que la lista está cargada, y que el algoritmo recibe como
parámetro de entrada la dirección del primer elemento.

```
accion ejercicio(prim: puntero a nodo) es
  ambiente
    nodo = registro
      dato: entero
      prox: puntero a nodo
    fin_registro

    prim, ant, p: puntero a nodo // <- lista original
    ant2, prim2, q: puntero a nodo // <- lista creada
  proceso
    p := prim
    prim2 := nil

    mientras (p <> nil) hacer
      si ((*p.dato mod 10) = 0) entonces

        // creamos nuevo nodo con dato de p
        nuevo(q)
        *q.dato := *p.dato
        *q.prox := nil

        // carga encolada, se añaden siempre al final
        si (prim2 = nil) entonces

          // si la lista está vacía
          prim2 := q
          ant2 := q
        contrario

          // si ya tiene nodos, lo añadimos al final
          *ant2.prox := q
          ant2 := q
        fin_si

        // eliminamos el nodo de la lista original
        si (p = prim) entonces // 1° caso: eliminar primer nodo
          prim := *p.prox
          disponer(p)
          p := prim
        contrario // 2° caso: eliminar nodo del medio
          *ant.prox := *p.prox
          disponer(p)
          p := *ant.prox
        fin_si
      contrario
        ant := p
        p := *p.prox
      fin_si
    fin_mientras
fin_accion
```
