## 4.06
en el restaurante ñoquis se está pensando en una solución informática para el
soporte de datos del nuevo sistema de atención a clientes. se han decidido por
listas por su dinamismo en cuanto a la cantidad de elementos. diseñe un
algoritmo que realice las siguientes funciones:

- añadir cliente al ser atendido (lista simple ordenada por nombre del cliente)
- registrar su consumo (acumular el total consumido en valores de montos)
- realizar el cobro (emitir ticket con nombre, fecha, número de mesa y total)
- eliminar del listado de atención

la información almacenada debe mantenerse ordenada por nombre del cliente.

```
accion ejercicio(prim: puntero a nodo) es
  ambiente
    cliente = registro
      nombre: an(30)
      fecha: an(10)
      mesa: entero
      total: real
    fin_registro

    nodo = registro
      dato: cliente
      prox: puntero a nodo
    fin_registro

    p, a, q: puntero a nodo
    nombre_buscado: an(30)
    monto: real
    opcion: entero
    continuar: caracter

    procedimiento agregar_cliente() es
      nuevo(q)
      escribir("ingrese el nombre del cliente: ")
      leer(*q.dato.nombre)
      escribir("ingrese la fecha: ")
      leer(*q.dato.fecha)
      escribir("ingrese el número de mesa: ")
      leer(*q.dato.mesa)
      *q.dato.total := 0

      p := prim
      a := nil
      mientras (p <> nil) y (*p.dato.nombre < *q.dato.nombre) hacer
        a := p
        p := *p.prox
      fin_mientras

      *q.prox := p
      si a = nil entonces
        prim := q
      contrario
        *a.prox := q
      fin_si
    fin_procedimiento

    procedimiento registrar_consumo() es
      escribir("ingrese el nombre del cliente: ")
      leer(nombre_buscado)
      escribir("ingrese el monto del consumo: ")
      leer(monto)

      p := prim
      mientras (p <> nil) y (*p.dato.nombre <> nombre_buscado) hacer
        p := *p.prox
      fin_mientras

      si p = nil entonces
        escribir("error: cliente no encontrado.")
      contrario
        *p.dato.total := *p.dato.total + monto
      fin_si
    fin_procedimiento

    procedimiento cobrar() es
      escribir("ingrese el nombre del cliente: ")
      leer(nombre_buscado)

      p := prim
      mientras (p <> nil) y (*p.dato.nombre <> nombre_buscado) hacer
        p := *p.prox
      fin_mientras

      si p = nil entonces
        escribir("error: cliente no encontrado.")
      contrario
        escribir("--- ticket ---")
        escribir("nombre: ", *p.dato.nombre)
        escribir("fecha: ", *p.dato.fecha)
        escribir("mesa: ", *p.dato.mesa)
        escribir("total: ", *p.dato.total)
      fin_si
    fin_procedimiento

    procedimiento eliminar_cliente() es
      escribir("ingrese el nombre del cliente: ")
      leer(nombre_buscado)

      p := prim
      a := nil
      mientras (p <> nil) y (*p.dato.nombre <> nombre_buscado) hacer
        a := p
        p := *p.prox
      fin_mientras

      si p = nil entonces
        escribir("error: cliente no encontrado.")
      contrario
        si a = nil entonces
          prim := *p.prox
        contrario
          *a.prox := *p.prox
        fin_si
        disponer(p)
      fin_si
    fin_procedimiento
  proceso
    repetir
      escribir("acciones")
      escribir("  [1] añadir cliente")
      escribir("  [2] registrar consumo")
      escribir("  [3] realizar cobro")
      escribir("  [4] eliminar cliente")
      escribir("ingrese la operación a realizar: ")
      leer(opcion)

      segun opcion hacer
        1: agregar_cliente()
        2: registrar_consumo()
        3: cobrar()
        4: eliminar_cliente()
        contrario: escribir("error: opción inválida.")
      fin_segun

      escribir("desea realizar otra operación? [s/n]")
      leer(continuar)
    hasta que continuar = 'n'
fin_accion
```
